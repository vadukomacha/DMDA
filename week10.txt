Time Series:

sales <- c(100,200,300,400,500,600,700,800,900,1000,1100,1200)
sales_ts <- ts(sales, start = c(2025,1), frequency = 12)
print(sales_ts)
plot(sales_ts, type = "o", main = "time series analysis", xlab = "months", ylab = "sales", lwd = 3)


Nonlinear least square analysis:

year = c(1,2,3,4,5,6)
population = c(100,110,125,140,160,185)
model = nls(population ~ a * exp(b * year), start = list(a = 90, b = 0.1))
print(model)
predicted = predict(model)
print(predicted)
plot(year, population, pch = 19, main = "population growth-non linear regression", xlab = "years", ylab = "population(thousands)")
lines(year, predicted, lwd = 2)
new_year = data.frame(year = 8)
predicted_8 = predict(model, newdata = new_year)
