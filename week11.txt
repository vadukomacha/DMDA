Normal Distribution

i.DNROM
x <- seq(-10, 10, by = 0.1)
y <- dnorm(x, mean = 2.5, sd = 0.5)
model=lm(y~x)
plot(x, y, main = "Normal Distribution", xlab = "X", ylab = "Density", type = "l")
abline(model,col="blue",lwd=2)

ii.PNROM:
x <- seq(-10, 10, by = 0.1)
y <- pnorm(x, mean = 2.5, sd = 0.5)
plot(x, y,main = "Cumulative Normal Distribution", xlab = "X",ylab = "Cumulative Probability", type = "l")

iii.QNROM
x <- seq(-0, 1, by = 0.02)
y <- qnorm(x, mean = 2, sd = 1)
plot(x, y,main = "Quantile Normal Distribution",xlab = "X",ylab = "Quantile Distribution",type = "l")

iv. RNORM
y <- rnorm(50)
hist(y,main = "Random Normal Distribution", xlab = "Values",col = "lightblue")



Binomial Distribution

i.DBINOM
x <- seq(0, 50, by = 1)
y <- dbinom(x, 50, 0.5)
plot(x, y,main = "Binomial Distribution",xlab = "Number of Successes", ylab = "Probability",type = "h")

ii.PBINOM
x <- 0:51
y <- pbinom(x, 51, 0.5)
plot(x, y,main = "Cumulative Binomial Distribution",xlab = "Number of Successes",ylab = "Cumulative Probability",type = "s")

iii.QBINOM
x=seq(0,1,by=0.02)
y=qbinom(x,2.0,0.5)
plot(x,y,main="Quantile Binomial Distribution",xlab="Number of Success",ylab="Quantile",type="s")

IV. RBINOM
y=rbinom(50,10,0.5)
hist(y,main="Random Binomial Distribution",xlab="values",col="blue")
