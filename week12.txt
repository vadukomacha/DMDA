t-test:

x=c(0.593,0.142,0.329,0.691,0.231,0.793,0.392,0.418) 
result=t.test(x,alternative="greater",mu=0.3) 
print(result) 
hist(x,probability = TRUE,main = "One-Sample t-Test",xlab = "Sample Values", ylab = "Density",col = "lightblue",border = "Green")


f-test:

x=c(66,67,75,76,82,84,88,90,92) 
y=c(64,66,74,78,82,85,87,92,93,95,97) 
result= var.test(x,y) 
print(result) 
boxplot(x, y,names = c("Group X", "Group Y"),main = "Comparison of Variances",xlab = "Groups",ylab = "Values",col = c("lightblue", "lightgreen"))
