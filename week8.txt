mean:

x=c(10,20,30,40,50)
result=mean(x)
print(result)


median:

x=c(10,20,-30,40,-50,-10,-20)
result=median(x)
print(result)


mode:

getmode=function(v)
{
  uniqv=unique(v)
  uniqv[which.max(tabulate(match(v,uniqv)))]
}
v=c(10,20,10,30,20,10,40)
result=getmode(v)
print(result)


covariance:

x=c(10,20,30,40)
y=c(50,60,70,80)
result=cov(x,y)
print(result)
boxplot(x,y)


regression:

x=c(25,35,45,60)
y=c(2,11,10,25)
result=lm(x~y)
print(result)
boxplot(x~y)
