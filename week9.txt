Linear Regression:

stdhrs<-c(1,2,3,4,5)
marks<-c(40,50,60,70,80)
student<-data.frame(stdhrs,marks)
print(student)
model=lm(marks~stdhrs,data=student)
plot(stdhrs,marks,model="linear regression",xlab="stdhrs",ylab="marks",pch=19,col="blue")
abline(model,col="purple",lwd=2)


Multiple Regression:

experience<-c(1,2,3,4,5,6)
education<-c(12,12,15,15,16,18)
salary<-c(25000,28000,35000,42000,50000,60000)
employee<-data.frame(experience,education,salary)
print(employee)
model<-lm(salary~experience+education,data=employee)
plot(employee$experience,employee$salary,main="Experience vs Salary",xlab="Experience(Years)",ylab="Salary",pch=19,col="blue")
abline(lm(salary~experience))


Logistic Regression:

students<-data.frame(studyhrs=c(1,2,3,4,5,6,7,8),pass=c(0,0,0,1,1,1,1,1))
model<-glm(pass~studyhrs,data=students,family=binomial)
summary(model)
students$Probability<-predict(model,type="response")
students$Prediction<-ifelse(students$Probability>=0.5,1,0)
students
plot(students$studyhrs,students$pass,pch=19,xlab="Study Hours",ylab="Probability of Passing",main="Logistic Regression")



Poisson Regression:

hours<-c(1,2,3,4,5,6,7,8)
questions<-c(2,3,4,6,8,10,13,16)
data<-data.frame(hours,questions)
print(data)
model<-glm(questions~hours,family=poisson,data=data)
summary(model)
predicted<-predict(model,type="response")
print(predicted)
plot(data$hours,data$questions,pch=19,xlab="Study Hours",ylab="Questions Answered",main="Poisson Regression")
lines(data$hours,predicted,lwd=2)
